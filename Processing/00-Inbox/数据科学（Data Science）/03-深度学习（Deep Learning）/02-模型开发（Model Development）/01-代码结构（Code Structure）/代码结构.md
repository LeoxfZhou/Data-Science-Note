### 最基础的深度学习代码

```Python
import os
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader
from torchvision import datasets, transforms

# ==========================================
# 第一步：定义最简单的数据预处理（无任何增强）
# ==========================================
def get_advanced_train_transform(): 
	return transforms.Compose([ 
		# 1. 强行转换成 RGB 三通道（防止有些图片是单通道黑白图或有透明通道导致报错） 
		transforms.Lambda(lambd=lambda img: img.convert("RGB")), 
		
		# 2. 【数据增强】50% 的概率让图片水平翻转（左变右，右变左） 
		transforms.RandomHorizontalFlip(p=0.5), 
		
		# 3. 【数据增强】50% 的概率让图片垂直翻转（倒过来） 
		transforms.RandomVerticalFlip(p=0.5), 
		
		# 4. 【数据增强】随机旋转图片（-10度 到 +10度 之间随机转） 
		transforms.RandomRotation(degrees=10), 
		
		# 5. 缩放到固定大小 
		transforms.Resize((224, 224)), 
		
		# 6. 转为 Tensor 并归一化到 [0, 1] 
		transforms.ToTensor() 
	])

# ==========================================
# 第二步：手写一个极其简单的 3 层卷积神经网络 (CNN)
# ==========================================
class Simple3LayerCNN(nn.Module):
    def __init__(self, num_classes):
        super().__init__()
        
        # 提取特征的第一层：输入是3通道(RGB)，输出16通道，卷积核大小3x3
        self.conv1 = nn.Conv2d(in_channels=3, out_channels=16, kernel_size=3, padding=1)
        self.relu1 = nn.ReLU() # 激活函数，增加非线性能力
        self.pool1 = nn.MaxPool2d(kernel_size=2, stride=2) # 池化层，图片尺寸减半：224 -> 112
        
        # 提取特征的第二层：输入是16通道，输出32通道
        self.conv2 = nn.Conv2d(in_channels=16, out_channels=32, kernel_size=3, padding=1)
        self.relu2 = nn.ReLU()
        self.pool2 = nn.MaxPool2d(kernel_size=2, stride=2) # 图片尺寸再次减半：112 -> 56
        
        # 第三层：全连接分类器（将二维特征图展平后，映射到最终的类别数）
        # 经过两次池化，图片变成了 56x56 大小，通道数是 32
        self.fc = nn.Linear(in_features=32 * 56 * 56, out_features=num_classes)

    def forward(self, x):
        # 数据流向：输入 x -> 卷积1 -> 激活 -> 池化1
        x = self.pool1(self.relu1(self.conv1(x)))
        # 数据流向：-> 卷积2 -> 激活 -> 池化2
        x = self.pool2(self.relu2(self.conv2(x)))
        
        # 展平操作（Flatten）：把 [Batch_Size, 32, 56, 56] 变成 [Batch_Size, 32*56*56]
        # 只有变成一维向量，全连接层(nn.Linear)才能接收
        x = x.view(x.size(0), -1)
        
        # 最终分类输出
        out = self.fc(x)
        return out

# ==========================================
# 第三步：极简训练主循环
# ==========================================
def mini_training_loop(train_dir, batch_size=8, epochs=3):
    # 自动选择设备：有显卡用 GPU(cuda)，没有就用 CPU
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"🔥 当前正在使用的设备: {device}")

    # 1. 使用 ImageFolder 自动根据文件夹名字切分标签并加载数据
    train_dataset = datasets.ImageFolder(root=train_dir, transform=get_minimal_transform())
    num_classes = len(train_dataset.classes) # 自动获取分类的总数（比如 17 种花）
    print(f"📦 成功加载数据集，样本总数: {len(train_dataset)}，总类别数: {num_classes}")

    # 2. 用 DataLoader 捆绑打包成一个一个 Batch
    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)

    # 3. 实例化我们刚才手写的极简网络，并推送到设备(CPU/GPU)中
    model = Simple3LayerCNN(num_classes=num_classes).to(device)

    # 4. 定义标准损失函数（多分类用交叉熵）和基本优化器（SGD）
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.SGD(model.parameters(), lr=0.01)

    # 5. 核心双重循环：外层循环控制刷几遍书 (Epoch)，内层循环控制一页一页看 (Batch)
    model.train() # 告诉模型：现在进入训练模式
    print("\n🚀 开始炼丹...\n")
    
    for epoch in range(epochs):
        running_loss = 0.0
        
        for batch_idx, (images, labels) in enumerate(train_loader):
            # 将数据和标签推送到同一个设备上
            images = images.to(device)
            labels = labels.to(device)

            # --- 核心五步法 ---
            optimizer.zero_grad()               # ① 梯度清零（防止上一批次的计算结果干扰现在）
            outputs = model(images)             # ② 前向传播（让模型猜答案）
            loss = criterion(outputs, labels)   # ③ 计算损失（对答案，算差距）
            loss.backward()                     # ④ 反向传播（顺藤摸瓜算导数，找出哪个参数该骂）
            optimizer.step()                    # ⑤ 更新参数（根据方向稍微修改权重）

            # 累计当前 Epoch 的总 Loss
            running_loss += loss.item()

            # 每看 2 个 Batch 打印一次进度
            if batch_idx % 2 == 0:
                print(f"Epoch [{epoch+1}/{epochs}] | Batch [{batch_idx}/{len(train_loader)}] | 当前 Batch Loss: {loss.item():.4f}")
        
        # 算完一遍数据集，打印一轮的平均 Loss
        epoch_loss = running_loss / len(train_loader)
        print(f"🌟 Epoch [{epoch+1}/{epochs}] 结束! 平均 Loss 为: {epoch_loss:.4f}\n")

    print("🎉 极简训练跑通！你已经成功迈出了深度学习的第一步！")

if __name__ == '__main__':
    # ⚠️ 请把这里的路径替换为你电脑里实际的 17flowers 训练集路径 ⚠️
    TRAIN_DIR = r"D:\datas\17flowers\train" 
    
    if os.path.exists(TRAIN_DIR):
        mini_training_loop(train_dir=TRAIN_DIR, batch_size=4, epochs=3)
    else:
        print(f"❌ 路径不存在: '{TRAIN_DIR}'，请修改代码最底部的 TRAIN_DIR 为你本地的真实路径！")
```
